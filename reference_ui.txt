// Generate volcano data
function genVolcano(cancer) {
  const up = MOCK[cancer].expression.top_upregulated;
  const dn = MOCK[cancer].expression.top_downregulated;
  const pts = [];
  for (let i = 0; i < 420; i++) {
    const lfc = (Math.random() - 0.5) * 9;
    const negLogP = Math.random() * 6 * (Math.abs(lfc) > 1.5 ? 1.4 : 0.6);
    const sig = Math.abs(lfc) > 1 && negLogP > 1.3;
    pts.push({ x: +lfc.toFixed(3), y: +negLogP.toFixed(3), gene: `G${i}`, sig, dir: lfc > 0 ? "up" : "dn" });
  }
  up.forEach(g => pts.push({ x: +(1.8 + Math.random() * 2.8).toFixed(3), y: +(2.5 + Math.random() * 5).toFixed(3), gene: g, sig: true, dir: "up", named: true }));
  dn.forEach(g => pts.push({ x: +(-1.8 - Math.random() * 2.8).toFixed(3), y: +(2.5 + Math.random() * 5).toFixed(3), gene: g, sig: true, dir: "dn", named: true }));
  return pts;
}

// TMB histogram
function genTMB(cancer) {
  const med = MOCK[cancer].mutation.median_tmb;
  const scale = cancer === "SKCM" ? 8 : 1;
  return Array.from({ length: 24 }, (_, i) => ({
    x: +(i * scale * 0.6).toFixed(1),
    y: Math.round(180 * Math.exp(-0.5 * Math.pow((i - 3) / 4, 2)) + Math.random() * 20)
  }));
}

// Network nodes for multi-omics canvas
function genNodes(cancer) {
  const int = MOCK[cancer].integration;
  const center = { id: "center", x: 50, y: 50, r: 28, type: "hub", label: "MULTI-OMICS", sublabel: "Convergence", color: "#06B6D4" };
  const nodes = [center];
  const addRing = (items, type, color, rx, ry, spread) => {
    items.forEach((gene, i) => {
      const angle = (i / items.length) * Math.PI * 2 - Math.PI / 2;
      nodes.push({ id: gene, x: rx + Math.cos(angle) * spread, y: ry + Math.sin(angle) * spread * 0.62, r: 16, type, label: gene, color });
    });
  };
  addRing(int.mutated_and_upregulated, "mut_up", "#10B981", 50, 50, 28);
  addRing(int.mutated_and_downregulated, "mut_dn", "#EF4444", 50, 50, 22);
  addRing(int.mutated_only.slice(0, 4), "mut", "#06B6D4", 50, 50, 38);
  addRing(int.deg_only.slice(0, 5), "deg", "#8B5CF6", 50, 50, 42);
  return nodes;
}

// ─── MULTI-OMICS NETWORK CANVAS ───────────────────────────────────────────────
function NetworkCanvas({ cancer, t, onGeneSelect, selectedGene }) {
  const nodes = useMemo(() => genNodes(cancer), [cancer]);
  const [tick, setTick] = useState(0);
  const [hoveredNode, setHoveredNode] = useState(null);
  const animRef = useRef();

  useEffect(() => {
    let frame;
    const animate = () => { setTick(k => k + 1); frame = requestAnimationFrame(animate); };
    frame = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(frame);
  }, []);

  const center = nodes[0];
  const others = nodes.slice(1);

  const TYPE_COLORS = {
    mut_up: t.emerald, mut_dn: t.red, mut: t.cyan, deg: t.violet, hub: t.cyan
  };

  const nodeJitter = (id, axis) => {
    const seed = id.charCodeAt(0) * 137.5 + id.charCodeAt(id.length - 1) * 41.3;
    const f = axis === "x" ? 0.8 : 1.2;
    return Math.sin(tick * 0.02 + seed) * f;
  };

  return (
    <svg viewBox="0 0 100 100" style={{ width: "100%", height: "100%", overflow: "visible" }} preserveAspectRatio="xMidYMid meet">
      <defs>
        <filter id="glow">
          <feGaussianBlur stdDeviation="1.5" result="coloredBlur" />
          <feMerge><feMergeNode in="coloredBlur" /><feMergeNode in="SourceGraphic" /></feMerge>
        </filter>
        <filter id="glowStrong">
          <feGaussianBlur stdDeviation="3" result="coloredBlur" />
          <feMerge><feMergeNode in="coloredBlur" /><feMergeNode in="SourceGraphic" /></feMerge>
        </filter>
        <radialGradient id="hubGrad" cx="50%" cy="50%" r="50%">
          <stop offset="0%" stopColor={t.cyan} stopOpacity="0.3" />
          <stop offset="100%" stopColor={t.cyan} stopOpacity="0" />
        </radialGradient>
      </defs>

      {/* Orbital rings */}
      {[22, 34, 44].map((r, i) => (
        <circle key={r} cx="50" cy="50" r={r} fill="none" stroke={t.cyan} strokeWidth="0.15" strokeDasharray={i === 0 ? "2 3" : i === 1 ? "1 4" : "0.5 6"} opacity={0.25}
          style={{ transformOrigin: "50px 50px", animation: `${i % 2 === 0 ? "rotateRing" : "rotateRingRev"} ${40 + i * 20}s linear infinite` }}
        />
      ))}

      {/* Connection lines */}
      {others.map((n, i) => {
        const nx = n.x + nodeJitter(n.id, "x");
        const ny = n.y + nodeJitter(n.id, "y");
        const isSelected = selectedGene && (n.id === selectedGene || center.id === selectedGene);
        const isHovered = hoveredNode === n.id;
        const col = TYPE_COLORS[n.type] || t.cyan;
        const opacity = selectedGene ? (n.id === selectedGene ? 0.9 : 0.1) : (isHovered ? 0.7 : 0.25);
        return (
          <line key={n.id} x1="50" y1="50" x2={nx} y2={ny}
            stroke={col} strokeWidth={isSelected || isHovered ? 0.5 : 0.2}
            opacity={opacity} strokeDasharray="2 2"
            style={{ animation: `networkPulse ${3 + i * 0.4}s ease-in-out infinite` }}
          />
        );
      })}

      {/* Hub center */}
      <circle cx="50" cy="50" r="12" fill="url(#hubGrad)" filter="url(#glowStrong)" />
      <circle cx="50" cy="50" r="10" fill={`${t.bg}CC`} stroke={t.cyan} strokeWidth="0.8" />
      <circle cx="50" cy="50" r="6" fill={`${t.cyan}44`} />
      <circle cx="50" cy="50" r="3" fill={t.cyan} filter="url(#glow)" />
      <text x="50" y="26" textAnchor="middle" fill={t.cyan} fontSize="2.2" fontWeight="700" fontFamily="'JetBrains Mono',monospace" opacity="0.8">MULTI-OMICS</text>
      <text x="50" y="29" textAnchor="middle" fill={t.textDim} fontSize="1.6" fontFamily="Inter" opacity="0.7">Convergence Hub</text>

      {/* Gene nodes */}
      {others.map((n, i) => {
        const nx = n.x + nodeJitter(n.id, "x");
        const ny = n.y + nodeJitter(n.id, "y");
        const col = TYPE_COLORS[n.type] || t.cyan;
        const isSelected = selectedGene === n.id;
        const isHovered = hoveredNode === n.id;
        const rScale = isSelected ? 1.3 : isHovered ? 1.15 : 1;
        const opacity = selectedGene ? (isSelected ? 1 : 0.3) : (isHovered ? 1 : 0.75);
        const r = (n.r / 10) * rScale;
        return (
          <g key={n.id} transform={`translate(${nx},${ny})`} onClick={() => onGeneSelect(n.id)} style={{ cursor: "pointer" }}
            onMouseEnter={() => setHoveredNode(n.id)} onMouseLeave={() => setHoveredNode(null)}>
            {isSelected && <circle cx="0" cy="0" r={r * 2.5} fill={col} opacity="0.08" style={{ animation: "ripple 2s ease-out infinite" }} />}
            <circle cx="0" cy="0" r={r * 1.8} fill={col} opacity="0.08" />
            <circle cx="0" cy="0" r={r} fill={`${t.bg}CC`} stroke={col} strokeWidth={isSelected ? 0.8 : 0.5} opacity={opacity} filter={isSelected || isHovered ? "url(#glow)" : "none"} style={{ transition: "r 0.2s, stroke-width 0.2s" }} />
            <circle cx="0" cy="0" r={r * 0.45} fill={col} opacity={opacity} style={{ animation: `float ${5 + i * 0.7}s ease-in-out infinite` }} />
            <text x="0" y={r + 2.2} textAnchor="middle" fill={isSelected ? col : t.textMid} fontSize="1.8" fontWeight={isSelected ? "700" : "500"} fontFamily="'JetBrains Mono',monospace" opacity={opacity} style={{ pointerEvents: "none" }}>{n.label}</text>
          </g>
        );
      })}

      {/* Legend */}
      {[{ type: "mut_up", label: "Mut + Up" }, { type: "mut_dn", label: "Mut + Down" }, { type: "mut", label: "Mutation" }, { type: "deg", label: "Expression" }].map(({ type, label }, i) => (
        <g key={type} transform={`translate(${1.5 + i * 22.5}, 97)`}>
          <circle cx="0" cy="-0.5" r="1.2" fill={TYPE_COLORS[type]} opacity="0.8" />
          <text x="2.2" y="0.5" fill={t.textDim} fontSize="1.8" fontFamily="Inter">{label}</text>
        </g>
      ))}
    </svg>
  );
}
